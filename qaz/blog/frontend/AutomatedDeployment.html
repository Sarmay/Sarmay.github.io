            # العاشقى قادام قوسۋ SSH تەرمينال 🚑

            1. cd جوبانى جٷكتەپ, ساقتاعىڭىز كەلەتٴىن كاتالوگقا ٶتٴىڭٴىز，جوبانى كٶشٴىرۋ

            2. سودان كەيٴىن cd كٶشٴىرٴىلگەن جوبانى ەنگٴىزٴىڭٴىز

            3. سودان كەيٴىن تەرٴىڭٴىز git config --global credential.helper store

            4. ورىندالعان git pullكودتى تارتىڭىز

            # ەكٴىنشٴى قادام قوسۋ WebHook قۇرال 📝

            1. قوسىڭىز, اتاۋ سٴىزدٴىڭ جوباڭىزدىڭ اعىلشىنشا مەكەن-جايىنا سٴايكەس كەلۋٴى كەرەك

            2. ستسەنارييدٴى ورىنداپ, تٶمەندەگٴى كودتى قويىڭىز دا, ٶزٴىڭٴىز ٶزگەرتٴىڭٴىز git اتىبٴىرگەجوبانى جٷكتەۋ كاتالوگى

              #!/bin/bash
echo ""
date --date='0 days ago' "+%Y-%m-%d %H:%M:%S"
echo "Start"
if [ ! -n "$1" ];
then
          echo "param erro"

          exit
fi
#جولدى ٶزٴىڭٴىز انىقتاڭىز

if [ -d "$gitPath" ]; then

        if [ ! -d ".git" ]; then

                echo "جاڭا جوبا كاتالوگى"

        fi

            3. كٴىلتتٴى كٶرۋ ٷشٴىن باسىڭىز

            # ٷشٴىنشٴى قادام قوسۋ git ✨

            1. ٶز جوباڭىزدى ەنگٴىزٴىپ, باسقارۋ تٷيمەسٴىن باسىڭىز，باسىڭىز webhook，ساقتاۋ ٷشٴىن مەكەنجايدى دٴال قازٴىر مەكەن-جايعا قويىڭىز

            # تٶرتٴىنشٴى قادام قوسۋ PM2 🚀

            1. PM2 ونىڭ جانىنداعى قالتا بەلگٴىشەسٴىن باسىڭىز ەنگٴىزۋpm2 تٷبٴىرلٴىك فايل جولى بولۋى كەرەك/www/server/panel/plugin/pm2

            2. ٶزگەرتۋ pm2_main.py فايل, جويۋ pm2_main.pyc فايل

               #!/usr/bin/python
   #coding: utf-8
   #-----------------------------
   # PM2
   #-----------------------------
   import sys,os,psutil
   os.chdir('/www/server/panel')
   sys.path.append("class/")
   import public,re,json

   class pm2_main:
       __SR = None
       __path = '/www/server/panel/plugin/pm2/list/'

       def __init__(self):
           s_path = '/www'
           if os.path.islink(s_path):
               s_path=os.readlink(s_path) + '/server'
           else:
               s_path += '/server'
               if os.path.islink(s_path):
                   s_path = os.readlink(s_path)
           self.__SR = '''#!/bin/bash
   PATH=/bin:/sbin:/usr/bin:/usr/sbin:/usr/local/bin:/usr/local/sbin:~/bin
   export PATH
   export HOME=/root
   export NVM_NODEJS_ORG_MIRROR=http://npm.taobao.org/mirrors/node
   export NVM_DIR="{}/nvm"
   source /www/server/nvm/nvm.sh ; '''.format(s_path)

       #تٴىزٴىم
       def List(self,get):
           try:
               tmp = public.ExecShell(self.__SR + "pm2 list -m|grep -v 'pm2 list'|grep -v 'In memory PM2'|grep -v 'Local PM2'")
               appList = tmp[0].strip().split('+---')
               result = []
               tmp = public.ExecShell('lsof -c node -P|grep LISTEN')
               plist = tmp[0].split('\n')
               for app in appList:
                   try:
                       if not app: continue
                       tmp2 = re.findall(r"([\w ]+):(.+)",app)
                       tmp3 = {}
                       for t in tmp2:
                           tmp3[t[0].strip()] = t[1].strip()
                       tmp2 = tmp3
                       appInfo = {}
                       appInfo['name'] = app.split('\n')[0].strip()
                       appInfo['id'] = tmp2['pm2 id']
                       appInfo['mode'] = tmp2['mode']
                       appInfo['pid'] = int(tmp2['pid'])
                       appInfo['status'] = tmp2['status']
                       appInfo['restart'] = tmp2['restarted']
                       appInfo['uptime'] = tmp2['uptime']
                       appInfo['cpu'] = 0
                       if appInfo['pid'] and appInfo['pid'] != 'N/A':
                           try:
                               appInfo['cpu'] = psutil.Process(appInfo['pid']).cpu_percent(0.3)
                           except:  appInfo['cpu'] = 0

                       appInfo['mem'] = tmp2['memory usage']
                       appInfo['user'] = 'root'
                       appInfo['watching'] = tmp2['watching']
                       appInfo['port'] = 'OFF'
                       appInfo['path'] = 'OFF'
                       for p in plist:
                           ptmp = p.split()
                           if len(ptmp) < 8: continue
                           if ptmp[1] == str(appInfo['pid']): appInfo['port'] = ptmp[8].split(':')[1].split('->')[0]

                       if os.path.exists(self.__path + appInfo['name']): appInfo['path'] = public.readFile(self.__path + appInfo['name'])
                       result.append(appInfo)
                   except: continue
               return result
           except:
               return public.returnMsg(False,'تەكسەرۋدٴى ٶتٴىنەمٴىز pm2 بۇل قالىپتى جاعداي ما!'+public.get_error_info())

       #ورناتىلعان كٴىتاپحانالاردى الىڭىز
       def GetMod(self,get):
           tmp = public.ExecShell(self.__SR + "npm list --depth=0 -global --json|grep -v '/www/server/nvm'")
           modList = json.loads(tmp[0])
           result = []
           for m in modList['dependencies'].keys():
               mod = {}
               mod['name'] = m
               mod['version'] = modList['dependencies'][m]['version']
               result.append(mod)
           return result

       #كٴىتاپحانانى ورناتىڭىز
       def InstallMod(self,get):
           os.system(self.__SR + 'npm install ' + get.mname + ' -g')
           return public.returnMsg(True,'ورناتۋ سٴاتتٴى ٶتتٴى!')

       #كٴىتاپحانانى جويۋ
       def UninstallMod(self,get):
           MyNot=['pm2','npm']
           if get.mname in MyNot: return public.returnMsg(False,'جويۋ مٷمكٴىن ەمەس['+get.mname+']')
           os.system(self.__SR + 'npm uninstall ' + get.mname + ' -g')
           return public.returnMsg(True,'سٴاتتٴى جويىلدى!')

       #الۋ Node نۇسقالار تٴىزٴىمٴى
       def Versions(self,get):
           result = {}
           rep = r'v\d+\.\d+\.\d+'
           tmp = public.ExecShell(self.__SR+'nvm ls-remote --lts|grep -v v0|grep -v iojs')
           result['list'] = re.findall(rep,tmp[0])
           tmp = public.ExecShell(self.__SR + "nvm version")
           result['version'] = tmp[0].strip()
           print(tmp)
           return result

       #اۋىستىرۋ Node نۇسقاسى
       def SetNodeVersion(self,get):
           version = get.version.replace('v','')
           estr = '''
   export NVM_NODEJS_ORG_MIRROR=http://npm.taobao.org/mirrors/node && nvm install %s
   nvm use --delete-prefix %s
   nvm alias default %s
   oldreg=`npm get registry`
   npm config set registry http://registry.npm.taobao.org/
   npm install pm2 -g
   npm config set registry $oldreg
   ''' % (version,version,version)
           os.system(self.__SR + estr)
           return public.returnMsg(True,'اۋىستىرىلدى['+get.version+']')

       #قوسۋ
       def Add(self,get):
           #get.pname = get.pname.encode('utf-8');
           if not re.match(r"^\w+$",get.pname):
               return public.returnMsg(False,'جوبا اتاۋىندا ارنايى بەلگٴىلەر بولماۋى كەرەك!')
           runFile = (get.path + '/' + get.run).replace('//','/')
           if not os.path.exists(runFile): return public.returnMsg(False,'كٶرسەتٴىلگەن فايل جوق!')
           Nlist = self.List(get)
           for node in Nlist:
               if get.pname == node['name']: return public.returnMsg(False,'كٶرسەتٴىلگەن جوبا اتاۋى بۇرىننان بار!')
           if os.path.exists(get.path + '/package.json') and not os.path.exists(get.path + '/package-lock.json'): os.system(self.__SR + "cd " + get.path + ' && npm install -s')
           os.system(self.__SR + 'cd '+get.path+' && pm2 start ' + runFile +  ' --name "'+get.pname+'" --watch|grep ' + get.pname)
           public.ExecShell(self.__SR + 'pm2 save && pm2 startup')
           if not os.path.exists(self.__path): os.system('mkdir -p ' + self.__path)
           public.writeFile(self.__path + get.pname,get.path)
           return public.returnMsg(True,'ADD_SUCCESS')

       #جٴىبەرۋ
       def Start(self,get):
           #get.pname = get.pname.encode('utf-8');
           result = public.ExecShell(self.__SR + 'pm2 start "' + get.pname + '" --watch|grep ' + get.pname)[0]
           if result.find('online') != -1: return public.returnMsg(True,'جوبا['+get.pname+']ٴىسكە قوسىلدى!')
           return public.returnMsg(False,'جوبا['+get.pname+']بەلسەندٴىرٴىلمەدٴى!')

       #توقتا
       def Stop(self,get):
           #get.pname = get.pname.encode('utf-8');
           result = public.ExecShell(self.__SR + 'pm2 stop "' + get.pname + '"|grep ' + get.pname)[0]
           if result.find('stoped') != -1: return public.returnMsg(True,'جوبا['+get.pname+']توقتادى!')
           return public.returnMsg(True,'جوبا['+get.pname+']توقتاتۋ سٴاتسٴىز اياقتالدى!')

       #قايتا جٷكتەڭٴىز
       def Restart(self,get):
           #get.pname = get.pname.encode('utf-8');
           result = public.ExecShell(self.__SR + 'pm2 restart "' + get.pname + '"')[0]
           if result.find('✓') != -1: return public.returnMsg(True,'جوبا['+get.pname+']قايتا ٴىسكە قوسىلدى!')
           return public.returnMsg(False,'جوبا['+get.pname+']قايتا ٴىسكە قوسۋ سٴاتسٴىز اياقتالدى!')

       #شامادان تىس جٷكتەمە
       def Reload(self,get):
           #get.pname = get.pname.encode('utf-8');
           result = public.ExecShell(self.__SR + 'pm2 reload "' + get.pname + '"')[0]
           if result.find('✓') != -1: return public.returnMsg(True,'جوبا['+get.pname+']قايتا جٷكتەلدٴى!')
           return public.returnMsg(False,'جوبا['+get.pname+']قايتا جٷكتەۋ سٴاتسٴىز اياقتالدى!')

       #جويۋ
       def Delete(self,get):
          # get.pname = get.pname.encode('utf-8');
           result = public.ExecShell(self.__SR + 'pm2 stop "'+get.pname+'" && pm2 delete "' + get.pname + '"|grep "' + get.pname+'"')[0]
           if result.find('✓') != -1:
               public.ExecShell(self.__SR + 'pm2 save && pm2 startup')
               if os.path.exists(self.__path + get.pname): os.remove(self.__path + get.pname)
               return public.returnMsg(True,'DEL_SUCCESS')
           return public.returnMsg(False,'DEL_ERROR')

       #جۋرنالدى الىڭىز
       def GetLogs(self,get):
           path = '/root/.pm2/pm2.log'
           if not os.path.exists(path): return 'قازٴىرگٴى ۋاقىتتا جۋرنالدار جوق'
           return public.readFile(path)

   if __name__ == "__main__":
       p = pm2_main()
       print(p.List(None))
